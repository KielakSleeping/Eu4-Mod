namespace = hussite_events

country_event = {
    id = kielak_hussite_events.1
    title = kielak_hussite_events.1.t
    desc = kielak_hussite_events.1.d
    picture = DEFENESTRATION_PRAGUE_eventPicture

    fire_only_once = yes

    trigger = {
        tag = VEN
        religion = catholic
        266 = { NOT = { owned_by = BOH } }
        prestige = 50
    }

    mean_time_to_happen = {
        months = 12
    }

    immediate = {
        set_country_flag = prague_fallen
    }

    option = {
        ai_chance = { factor = 100 }
        name = kielak_hussite_events.1.a
        if = {
            limit = {
                has_estate = estate_church
            }
            add_estate_loyalty = {
                estate = estate_church 
                loyalty = -10
            }
        }           
        add_country_modifier = {
            name = "hussite_refugees"
            duration = 3650
        }
        set_country_flag = accepted_hussite_refugees        
    }

    option = {
        ai_chance = { factor = 1 }
        name = kielak_hussite_events.1.b
        add_country_modifier = {
            name = "denied_hussite_refugees"
            duration = 3650
        }
        if = {
            limit = {
                has_estate = estate_church
            }
            add_estate_loyalty = {
                estate = estate_church 
                loyalty = 10
            }
        }
        set_country_flag = denied_hussite_refugees
    }

}
